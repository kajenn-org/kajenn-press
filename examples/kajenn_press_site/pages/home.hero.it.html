def build(source, data):
    doc = source.html(lang="en")
    doc.head().title("^page.title")
    body = doc.body()
    body.nav().a("^texts.nav_home", href="^texts.href_home")
    body.div(node_label="body", kpress_slot="body")

# Il template costruisce il layout una volta.
# Ogni pagina riempie i suoi slot con i contenuti.
# Publish produce HTML statico per lingua.
